import { Component, input } from '@angular/core';
import { HandHistoryItem } from '@hbg/shared-models';
import { getTileAssetPath } from '../tile-asset-path';

/**
 * Displays the recent hand-history entries shown in the game sidebar.
 */
@Component({
  selector: 'lib-hand-history',
  templateUrl: './hand-history.html',
  styleUrls: ['../game-ui.animation-tokens.css', './hand-history.css'],
  standalone: true
})
export class HandHistory {
  history = input.required<HandHistoryItem[]>();

  tileAssetPath = getTileAssetPath;

  tileBackgroundImage(tileIdPath: string): string {
    return `url("${tileIdPath}")`;
  }
}

<div class="flex flex-col gap-2">
  <span class="sticky top-0 block font-mono text-[0.52rem] tracking-[0.22em] uppercase text-c-muted text-center bg-c-bg pb-1 z-10">HISTORY</span>

  @if (history().length === 0) {
    <span class="font-mono text-[0.6rem] text-c-muted tracking-[0.06em]">No hands played yet</span>
  } @else {
    <div class="flex flex-col">
      @for (row of history(); track row.round) {
        <div
          class="hh-row flex flex-col gap-1.5 py-1.5 border-b border-c-indigo/10 last:border-b-0"
          [class.hh-row--win]="row.result === 'win'"
          [class.hh-row--lose]="row.result === 'lose'"
        >
          <div class="flex items-center gap-1.5">
            <span class="font-mono text-[0.58rem] text-c-muted w-5 shrink-0">R{{ row.round }}</span>
            <span class="font-mono text-[0.5rem] text-c-muted tracking-[0.06em] flex-1">{{ row.bet.toUpperCase() }}</span>
            <span
              class="hh-result font-mono text-[0.5rem] font-bold tracking-[0.12em]"
              [class.hh-result--win]="row.result === 'win'"
              [class.hh-result--lose]="row.result === 'lose'"
            >{{ row.result === 'win' ? 'WON' : 'LOST' }}</span>
            <span
              class="font-mono text-[0.68rem] font-bold"
              [style.color]="row.scoreChange > 0 ? 'var(--c-green)' : 'var(--c-pink)'"
            >{{ row.scoreChange > 0 ? '+' : '' }}{{ row.scoreChange }}</span>
          </div>

          @if (row.visibleHand || row.hiddenHand) {
            <div class="hh-hands">
              @if (row.hiddenHand) {
                <div class="hh-mini-hand-row">
                  <div class="hh-mini-hand">
                    @for (tile of row.hiddenHand.tiles; track tile.id) {
                      <span
                        class="hh-mini-tile-frame"
                        aria-hidden="true"
                        [style.background-image]="tileBackgroundImage(tileAssetPath(tile))"
                      ></span>
                    }
                  </div>
                  <span class="hh-mini-total">{{ row.hiddenHand.total }}</span>
                </div>
              }
              @if (row.visibleHand) {
                <div class="hh-mini-hand-row">
                  <div class="hh-mini-hand">
                    @for (tile of row.visibleHand.tiles; track tile.id) {
                      <span
                        class="hh-mini-tile-frame"
                        aria-hidden="true"
                        [style.background-image]="tileBackgroundImage(tileAssetPath(tile))"
                      ></span>
                    }
                  </div>
                  <span class="hh-mini-total">{{ row.visibleHand.total }}</span>
                </div>
              }
            </div>
          }
        </div>
      }
    </div>
  }
</div>
